  endif()
  if(FEDSPATIAL_SCALAPACK_PROVIDER STREQUAL "CRAY_LIBSCI")
    find_path(FEDSPATIAL_LIBSCI_INCLUDE cblas.h
      HINTS "$ENV{CRAY_LIBSCI_PREFIX_DIR}/include" REQUIRED)
    find_library(FEDSPATIAL_LIBSCI_MPI NAMES sci_cray_mpi_mp sci_gnu_mpi_mp
      HINTS "$ENV{CRAY_LIBSCI_PREFIX_DIR}/lib" REQUIRED)
    find_library(FEDSPATIAL_LIBSCI_LOCAL NAMES sci_cray_mp sci_gnu_mp
      HINTS "$ENV{CRAY_LIBSCI_PREFIX_DIR}/lib" REQUIRED)
    target_include_directories(fedspatial_cpu PUBLIC ${FEDSPATIAL_LIBSCI_INCLUDE})
    target_link_libraries(fedspatial_cpu PUBLIC
      ${FEDSPATIAL_LIBSCI_MPI} ${FEDSPATIAL_LIBSCI_LOCAL})
    target_compile_definitions(fedspatial_cpu PUBLIC
      FEDSPATIAL_CRAY_LIBSCI FEDSPATIAL_BLAS_LOCAL)
  elseif(FEDSPATIAL_SCALAPACK_PROVIDER STREQUAL "MKL")
    find_path(FEDSPATIAL_MKL_INCLUDE mkl_scalapack.h HINTS "$ENV{MKLROOT}/include" REQUIRED)
    foreach(component scalapack_lp64 blacs_openmpi_lp64 intel_lp64 gnu_thread core)
      find_library(FEDSPATIAL_MKL_${component} NAMES mkl_${component}
        HINTS "$ENV{MKLROOT}/lib/intel64" REQUIRED)
      target_link_libraries(fedspatial_cpu PUBLIC ${FEDSPATIAL_MKL_${component}})
    endforeach()
    target_include_directories(fedspatial_cpu PUBLIC ${FEDSPATIAL_MKL_INCLUDE})
    target_compile_definitions(fedspatial_cpu PUBLIC
      FEDSPATIAL_MKL_LOCAL FEDSPATIAL_BLAS_LOCAL)
  else()
    message(FATAL_ERROR "Unknown FEDSPATIAL_SCALAPACK_PROVIDER=${FEDSPATIAL_SCALAPACK_PROVIDER}")
  endif()
  target_sources(fedspatial_cpu PRIVATE src/cpu_distributed.cpp)
  # The same oneMKL link line also provides the local CBLAS/LAPACKE kernels.
  # Use them when a worker has one rank instead of falling back to the
  # handwritten O(n^3) loops in sync_cpu.cpp.
  target_compile_definitions(fedspatial_cpu PUBLIC FEDSPATIAL_SCALAPACK)

      add_test(NAME cpu_distributed_backend_blocks COMMAND ${MPIEXEC_EXECUTABLE}
